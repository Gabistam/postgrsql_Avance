Écrivez la procédure recompenser(pseudo, niveaux) qui fait gagner des niveaux à un aventurier et lui verse 50 pièces par niveau gagné. Elle lève une erreur si l'aventurier n'existe pas. Récompensez Kairo de 2 niveaux.

CREATE OR REPLACE PROCEDURE recompenser(p_pseudo TEXT, p_niveaux INTEGER)
LANGUAGE plpgsql
AS $$
BEGIN
    UPDATE aventuriers
    SET niveau = niveau + p_niveaux,
        pieces = pieces + 50 * p_niveaux
    WHERE pseudo = p_pseudo;
    IF NOT FOUND THEN
        RAISE EXCEPTION 'Aventurier inconnu : %', p_pseudo;
    END IF;
END;
$$;

26.2Revendre un objet

Écrivez la procédure vendre(pseudo, objet, INOUT gain) : l'aventurier revend un exemplaire d'un objet de son inventaire à la moitié de son prix. L'objet retourne en boutique, les pièces sont créditées, et la ligne d'inventaire disparaît s'il n'en reste plus. Zed revend son épée courte.

CREATE OR REPLACE PROCEDURE vendre(p_pseudo TEXT, p_objet TEXT, INOUT p_gain INTEGER DEFAULT NULL)
LANGUAGE plpgsql
AS $$
DECLARE
    v_av_id  INTEGER;
    v_obj    objets%ROWTYPE;
    v_qte    INTEGER;
BEGIN
    SELECT id INTO v_av_id FROM aventuriers WHERE pseudo = p_pseudo;
    IF NOT FOUND THEN RAISE EXCEPTION 'Aventurier inconnu : %', p_pseudo; END IF;
    SELECT * INTO v_obj FROM objets WHERE nom = p_objet;
    IF NOT FOUND THEN RAISE EXCEPTION 'Objet inconnu : %', p_objet; END IF;

    SELECT quantite INTO v_qte FROM inventaire
    WHERE aventurier_id = v_av_id AND objet_id = v_obj.id;
    IF NOT FOUND THEN
        RAISE EXCEPTION '% ne possède pas « % »', p_pseudo, p_objet
              USING HINT = 'Consultez votre inventaire avant de vendre.';
    END IF;

    p_gain := v_obj.prix / 2;
    IF v_qte = 1 THEN
        DELETE FROM inventaire WHERE aventurier_id = v_av_id AND objet_id = v_obj.id;
    ELSE
        UPDATE inventaire SET quantite = quantite - 1
        WHERE aventurier_id = v_av_id AND objet_id = v_obj.id;
    END IF;
    UPDATE objets      SET stock  = stock  + 1      WHERE id = v_obj.id;
    UPDATE aventuriers SET pieces = pieces + p_gain WHERE id = v_av_id;
END;
$$;
SQL
CALL vendre('zed', 'Épée courte');

CREATE OR REPLACE PROCEDURE inscrire_lot(p_pseudos TEXT[])
LANGUAGE plpgsql
AS $$
DECLARE
    v_pseudo TEXT;
BEGIN
    FOREACH v_pseudo IN ARRAY p_pseudos LOOP
        BEGIN                                    -- un bloc par pseudo : une erreur n'annule que lui
            INSERT INTO aventuriers (pseudo) VALUES (v_pseudo);
            RAISE NOTICE 'Inscrit : %', v_pseudo;
        EXCEPTION
            WHEN unique_violation THEN
                RAISE NOTICE 'Déjà pris : %', v_pseudo;
        END;
    END LOOP;
END;
$$;

CREATE OR REPLACE PROCEDURE reinitialiser_stocks(p_valeur INTEGER, p_lot INTEGER)
LANGUAGE plpgsql
AS $$
DECLARE
    r    RECORD;
    v_n  INTEGER := 0;
BEGIN
    FOR r IN SELECT id FROM objets ORDER BY id LOOP
        UPDATE objets SET stock = p_valeur WHERE id = r.id;
        v_n := v_n + 1;
        IF v_n % p_lot = 0 THEN
            COMMIT;
            RAISE NOTICE 'Lot validé : % objets', v_n;
        END IF;
    END LOOP;
    COMMIT;
END;
$$;
SQL
CALL reinitialiser_stocks(20, 4);

CREATE OR REPLACE FUNCTION trg_nom_objet()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    NEW.nom := regexp_replace(TRIM(NEW.nom), '\s+', ' ', 'g');
    RETURN NEW;
END;
$$;

CREATE TRIGGER nom_objet
BEFORE INSERT OR UPDATE OF nom ON objets
FOR EACH ROW
EXECUTE FUNCTION trg_nom_objet();

29.2Le trigger espion

Posez sur inventaire un trigger BEFORE qui affiche, pour chaque ligne touchée, TG_WHEN, TG_OP et TG_TABLE_NAME. Lancez une mise à jour puis une suppression, observez les messages, puis supprimez le trigger.

💡 Indice

Dans un trigger BEFORE DELETE, renvoyer NEW (qui vaut NULL) annulerait silencieusement la suppression.

▶ Résultat attendu 0 ligne
💬 Messages attendus
NOTICE:  BEFORE UPDATE sur inventaire
NOTICE:  BEFORE DELETE sur inventaire
✅ Corrigé
SQL
CREATE OR REPLACE FUNCTION trg_espion()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    RAISE NOTICE '% % sur %', TG_WHEN, TG_OP, TG_TABLE_NAME;
    IF TG_OP = 'DELETE' THEN
        RETURN OLD;            -- en DELETE, NEW est NULL : renvoyer OLD pour ne pas annuler
    END IF;
    RETURN NEW;
END;
$$;

CREATE TRIGGER espion
BEFORE INSERT OR UPDATE OR DELETE ON inventaire
FOR EACH ROW
EXECUTE FUNCTION trg_espion();
SQL

DELETE FROM inventaire WHERE aventurier_id = 8;

DROP TRIGGER espion ON inventaire;

💡 Indice

✖ Erreur attendue
ERROR:  Un niveau ne peut pas baisser : 15 → 10 pour mo
✅ Corrigé
SQL
CREATE OR REPLACE FUNCTION trg_niveau()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    IF NEW.niveau < OLD.niveau THEN
        RAISE EXCEPTION 'Un niveau ne peut pas baisser : % → % pour %',
              OLD.niveau, NEW.niveau, OLD.pseudo;
    END IF;
    RETURN NEW;
END;
$$;

CREATE TRIGGER niveau_croissant
BEFORE UPDATE OF niveau ON aventuriers
FOR EACH ROW
EXECUTE FUNCTION trg_niveau();

CREATE OR REPLACE FUNCTION trg_prix_achat()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    IF NEW.prix_total IS NULL THEN
        SELECT prix * NEW.quantite INTO NEW.prix_total
        FROM objets WHERE id = NEW.objet_id;
    END IF;
    RETURN NEW;
END;
$$;

RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    INSERT INTO stats_ventes (objet_id, quantite_vendue)

✅ Corrigé
SQL
CREATE OR REPLACE FUNCTION trg_nettoyage()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    INSERT INTO alertes (message)
    SELECT format('Nettoyage : %s ligne(s) d''inventaire supprimée(s)', COUNT(*))
    FROM supprimees
    HAVING COUNT(*) > 0;
    RETURN NULL;
END;
$$;

CREATE TRIGGER nettoyage
AFTER DELETE ON inventaire
REFERENCING OLD TABLE AS supprimees
FOR EACH STATEMENT
EXECUTE FUNCTION trg_nettoyage();

Chargez filmbox-s4.sql. Écrivez noter(pseudo, titre, note, INOUT moyenne) : elle vérifie la note (de 0,5 à 5, par demi-point), le membre et le film ; enregistre ou remplace la note ; ajoute un visionnage au journal ; renvoie la nouvelle moyenne du film. lea.reel donne 3,5 à Inception.

CREATE OR REPLACE PROCEDURE noter(p_pseudo TEXT, p_titre TEXT, p_note NUMERIC,
                                  INOUT p_moyenne NUMERIC DEFAULT NULL)
LANGUAGE plpgsql
AS $$
DECLARE
    v_user  INTEGER;
    v_film  INTEGER;
BEGIN
    IF p_note NOT BETWEEN 0.5 AND 5 OR p_note * 2 <> TRUNC(p_note * 2) THEN
        RAISE EXCEPTION 'Note invalide : % (de 0,5 à 5, par demi-point)', p_note;
    END IF;
    SELECT id INTO v_user FROM utilisateurs WHERE pseudo = p_pseudo;
    IF NOT FOUND THEN RAISE EXCEPTION 'Membre inconnu : %', p_pseudo; END IF;
    SELECT id INTO v_film FROM films WHERE titre = p_titre;
    IF NOT FOUND THEN RAISE EXCEPTION 'Film inconnu : %', p_titre; END IF;

    INSERT INTO notes (utilisateur_id, film_id, note, note_le)
    VALUES (v_user, v_film, p_note, CURRENT_DATE)
    ON CONFLICT (utilisateur_id, film_id)
        DO UPDATE SET note = EXCLUDED.note, note_le = EXCLUDED.note_le;
    INSERT INTO journal (utilisateur_id, film_id, date_visionnage)
    VALUES (v_user, v_film, CURRENT_DATE);

    SELECT ROUND(AVG(note), 2) INTO p_moyenne FROM notes WHERE film_id = v_film;
END;
$$;
SQL

CREATE OR REPLACE FUNCTION trg_films_stats()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
DECLARE
    v_film INTEGER := CASE WHEN TG_OP = 'DELETE' THEN OLD.film_id ELSE NEW.film_id END;
BEGIN
    INSERT INTO films_stats (film_id, nb_notes, moyenne)
    SELECT v_film, COUNT(*), COALESCE(ROUND(AVG(note), 2), 0)
    FROM notes WHERE film_id = v_film
    ON CONFLICT (film_id)
        DO UPDATE SET nb_notes = EXCLUDED.nb_notes, moyenne = EXCLUDED.moyenne;
    RETURN NULL;
END;
$$;

CREATE TRIGGER films_stats
AFTER INSERT OR UPDATE OR DELETE ON notes
FOR EACH ROW
EXECUTE FUNCTION trg_films_stats();

RETURNS TRIGGER
